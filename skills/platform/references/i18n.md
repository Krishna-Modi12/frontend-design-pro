### src/i18n.ts

```typescript
import { getRequestConfig } from 'next-intl/server';
import { notFound } from 'next/navigation';

export const locales = ['en', 'de', 'ja'] as const;
export type Locale = (typeof locales)[number];

export const defaultLocale: Locale = 'en';

export default getRequestConfig(async ({ locale }) => {
  // Validate that the incoming locale is supported
  if (!locales.includes(locale as Locale)) notFound();

  return {
    messages: (await import(`../messages/${locale}.json`)).default,
    // Optional: set timezone and now for consistent server/client rendering
    timeZone: 'UTC',
    now: new Date(),
  };
});
```

## 2. Middleware Setup

### src/middleware.ts

```typescript
import createMiddleware from 'next-intl/middleware';
import { locales, defaultLocale } from './i18n';

export default createMiddleware({
  // A list of all locales that are supported
  locales,

  // Used when no locale matches
  defaultLocale,

  // Locale prefix strategy:
  // 'always'    → /en/about, /de/about (every URL has prefix)
  // 'as-needed' → /about (default locale has no prefix), /de/about
  // 'never'     → /about (no prefixes; locale via Accept-Language only)
  localePrefix: 'always',

  // Automatically detect locale from Accept-Language header
  localeDetection: true,
});

export const config = {
  // Match all pathnames except for internal Next.js paths and static files
  matcher: [
    '/((?!_next|_vercel|.*\\..*).*)',
    '/',
  ],
};
```

### Locale prefix strategy reference

| Strategy     | Default locale URL | Other locale URL  | Use case                              |
|--------------|-------------------|-------------------|---------------------------------------|
| `'always'`   | `/en/about`       | `/de/about`       | Explicit, SEO-friendly, canonical     |
| `'as-needed'`| `/about`          | `/de/about`       | Clean default URL, localized others   |
| `'never'`    | `/about`          | `/about`          | Locale from cookie/header only        |

### Middleware with cookie-based locale persistence

```typescript
import createMiddleware from 'next-intl/middleware';
import { NextRequest } from 'next/server';
import { locales, defaultLocale } from './i18n';

const intlMiddleware = createMiddleware({
  locales,
  defaultLocale,
  localePrefix: 'as-needed',
});

  return intlMiddleware(request);
}

export const config = {
  matcher: ['/((?!api|_next|_vercel|.*\\..*).*)'],
};
```

## 4. useTranslations in Client Components

```tsx
'use client';

import { useTranslations } from 'next-intl';

export function WelcomeBanner({ name }: { name: string }) {
  // Namespace scopes all keys: t('home') → nav.home
  const t = useTranslations('nav');
  const tCommon = useTranslations('common');

  return (
    <div>
      {/* Basic key */}
      <h1>{t('home')}</h1>

      {/* Nested keys (dot notation within namespace) */}
      <nav>
        <span>{t('home')}</span>
        <span>{t('about')}</span>
      </nav>

      {/* Interpolation */}
      <p>{tCommon('greeting', { name })}</p>

      {/* Pluralization — ICU format in messages/en.json:
          "itemCount": "{count, plural, one {# item} other {# items}}" */}
      <p>{tCommon('itemCount', { count: 5 })}</p>
      {/* → "5 items" */}

      {/* Rich text with inline components */}
      <p>
        {tCommon.rich('terms', {
          link: (chunks) => (
            <a href="/terms" className="underline text-blue-600">
              {chunks}
            </a>
          ),
        })}
      </p>
    </div>
  );
}
```

## 6. Locale Switcher Component

```tsx
// src/components/LocaleSwitcher.tsx
'use client';

import { useLocale, useTranslations } from 'next-intl';
import { useRouter, usePathname } from '@/navigation';
import { locales, type Locale } from '@/i18n';
import { useTransition } from 'react';
import { useSearchParams } from 'next/navigation';

// Human-readable locale names using the Intl API
function getLocaleDisplayName(locale: Locale, displayLocale: Locale): string {
  return new Intl.DisplayNames([displayLocale], { type: 'language' }).of(locale) ?? locale;
}

export function LocaleSwitcher() {
  const currentLocale = useLocale() as Locale;
  const router = useRouter();
  const pathname = usePathname();
  const searchParams = useSearchParams();
  const [isPending, startTransition] = useTransition();

  function handleLocaleChange(nextLocale: Locale) {
    if (nextLocale === currentLocale) return;

    startTransition(() => {
      // Preserve current path and query params in the new locale
      const query = searchParams.toString();
      const pathWithQuery = query ? `${pathname}?${query}` : pathname;

      router.replace(pathWithQuery as any, { locale: nextLocale });
    });
  }

  return (
    <div className="relative inline-block">
      <select
        value={currentLocale}
        onChange={(e) => handleLocaleChange(e.target.value as Locale)}
        disabled={isPending}
        aria-label="Select language"
        className="
          appearance-none
          rounded-md border border-gray-300
          bg-white px-3 py-2 pr-8
          text-sm font-medium text-gray-700
          shadow-sm
          focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500
          disabled:opacity-50
          dark:bg-gray-800 dark:border-gray-600 dark:text-gray-200
        "
      >
        {locales.map((locale) => (
          <option key={locale} value={locale}>
            {getLocaleDisplayName(locale, currentLocale)}
          </option>
        ))}
      </select>

      {isPending && (
        <span className="absolute right-2 top-1/2 -translate-y-1/2 animate-spin">
          ⟳
        </span>
      )}
    </div>
  );
}
```

```tsx
import { useLocale } from 'next-intl';

          onClick={() =>

              : 'opacity-60 hover:opacity-100'
            }
          `}
        >
          {localeFlags[locale]}
        </button>
      ))}
    </div>
  );
```

### useFormatter hook (client components)

```tsx
'use client';

import { useFormatter, useNow } from 'next-intl';

export function FormattingExamples() {
  const format = useFormatter();
  const now = useNow({ updateInterval: 1000 }); // live-updating "now"

    <ul className="space-y-2 text-sm">
      {/* Date formatting */}
      <li>
        <strong>Short date:</strong>{' '}
        {format.dateTime(date, { dateStyle: 'short' })}
        {/* en → "6/15/25", de → "15.6.25", ja → "2025/06/15" */}
      </li>

      <li>
        <strong>Full date + time:</strong>{' '}
        {format.dateTime(date, {
          dateStyle: 'full',
          timeStyle: 'short',
        })}
        {/* en → "Sunday, June 15, 2025 at 2:30 PM" */}
      </li>

      <li>
        <strong>Custom format:</strong>{' '}
        {format.dateTime(date, {
          year: 'numeric',
          month: 'long',
          day: '2-digit',
          weekday: 'short',
        })}
      </li>

      {/* Number formatting */}
      <li>
        <strong>Plain number:</strong>{' '}
        {format.number(1234567.89)}
        {/* en → "1,234,567.89", de → "1.234.567,89", ja → "1,234,567.89" */}
      </li>

      <li>
        <strong>Percent:</strong>{' '}
        {format.number(0.8567, { style: 'percent', maximumFractionDigits: 1 })}
        {/* en → "85.7%", de → "85,7 %" */}
      </li>

      {/* Currency formatting */}
      <li>
        <strong>USD:</strong>{' '}
        {format.number(price, { style: 'currency', currency: 'USD' })}
        {/* en → "$1,234,567.89", de → "1.234.567,89 $" */}
      </li>

      <li>
        <strong>EUR:</strong>{' '}
        {format.number(price, {
          style: 'currency',
          currency: 'EUR',
          currencyDisplay: 'name',
        })}
        {/* en → "1,234,567.89 euros" */}
      </li>

      <li>
        <strong>JPY (no decimals):</strong>{' '}
        {format.number(price, {
          style: 'currency',
          currency: 'JPY',
          maximumFractionDigits: 0,
        })}
        {/* ja → "¥1,234,568" */}
      </li>

      {/* Relative time */}
      <li>
        <strong>Relative time:</strong>{' '}
        {format.relativeTime(pastDate, now)}
        {/* en → "45 minutes ago", de → "vor 45 Minuten" */}
      </li>

      <li>
        <strong>Relative time (future):</strong>{' '}
        {format.relativeTime(new Date(Date.now() + 1000 * 60 * 60 * 3), now)}
        {/* en → "in 3 hours" */}
      </li>
    </ul>
  );
}
```

## 8. Pluralization + ICU Message Format

### ICU message syntax in JSON

```json
{
  "items": {
    "count": "{count, plural, =0 {No items} one {# item} other {# items}}",
    "countWithType": "{count, plural, one {# {type}} other {# {type}s}}",

    "ordinal": "{position, selectordinal, one {#st} two {#nd} few {#rd} other {#th}} place",

    "gender": "{gender, select, male {He replied} female {She replied} other {They replied}}",

    "nested": "{itemCount, plural, =0 {No items in {folderCount, plural, one {# folder} other {# folders}}} one {1 item in {folderCount, plural, one {# folder} other {# folders}}} other {# items in {folderCount, plural, one {# folder} other {# folders}}}}",

    "escaped": "Price: {price} (use ''{literal}'' for literal braces)"
  }
}
```

### Using ICU messages in components

```tsx
'use client';

import { useTranslations } from 'next-intl';

export function ICUExamples({ count, position, gender }: {
  count: number;
  position: number;
  gender: 'male' | 'female' | 'other';
}) {
  const t = useTranslations('items');

  return (
    <ul>
      {/* Plural */}
      <li>{t('count', { count })}</li>
      {/* count=0 → "No items", count=1 → "1 item", count=5 → "5 items" */}

      {/* Selectordinal */}
      <li>{t('ordinal', { position })}</li>
      {/* position=1 → "1st place", position=2 → "2nd place", position=11 → "11th place" */}

      {/* Gender select */}
      <li>{t('gender', { gender })}</li>
      {/* gender="female" → "She replied" */}

      {/* Nested plural */}
      <li>{t('nested', { itemCount: 3, folderCount: 2 })}</li>
      {/* → "3 items in 2 folders" */}
    </ul>
  );
}
```

## 9. RTL Layout Support

### Detecting RTL locales

```typescript
// src/lib/rtl.ts
export const RTL_LOCALES = new Set(['ar', 'he', 'fa', 'ur']);

export function isRTL(locale: string): boolean {
  return RTL_LOCALES.has(locale);
}

// Or rely on Intl for robustness (requires 'unicode-bidi-embedding' support)
export function getTextDirection(locale: string): 'ltr' | 'rtl' {
  // Intl.Locale is available in Node 18+ and modern browsers
  try {
    const loc = new Intl.Locale(locale);
    // textInfo is stage-3 but widely supported
    return (loc as any).textInfo?.direction ?? (RTL_LOCALES.has(locale) ? 'rtl' : 'ltr');
  } catch {
    return RTL_LOCALES.has(locale) ? 'rtl' : 'ltr';
  }
}
```

```typescript
// src/lib/translate.ts
import { createTranslator } from 'next-intl';
import type { Messages } from 'next-intl';

// Use outside React components (e.g., in API routes, server utilities)
export async function getTranslator(locale: string, namespace: keyof Messages) {
  const messages = (await import(`../../messages/${locale}.json`)).default;

  return createTranslator({
    locale,
    messages,
    namespace,
  });
}
```

## 11. Testing i18n Components

### Setup: NextIntlClientProvider in tests

```tsx
// src/test-utils/i18n.tsx
import { NextIntlClientProvider } from 'next-intl';
import { render, type RenderOptions } from '@testing-library/react';
import type { ReactElement } from 'react';
import en from '../../messages/en.json';

interface I18nWrapperProps {
  locale?: string;
  messages?: Record<string, unknown>;
  children: React.ReactNode;
}

function I18nWrapper({ locale = 'en', messages = en, children }: I18nWrapperProps) {
  return (
    <NextIntlClientProvider locale={locale} messages={messages}>
      {children}
    </NextIntlClientProvider>
  );
}

// Custom render with i18n context
export function renderWithI18n(
  ui: ReactElement,
  options: { locale?: string; messages?: Record<string, unknown> } & Omit<RenderOptions, 'wrapper'> = {}
) {
  const { locale = 'en', messages = en, ...renderOptions } = options;

  return render(ui, {
    wrapper: ({ children }) => (
      <I18nWrapper locale={locale} messages={messages}>
        {children}
      </I18nWrapper>
    ),
    ...renderOptions,
  });
}
```

### Writing tests

```tsx
// src/components/__tests__/WelcomeBanner.test.tsx
import { renderWithI18n } from '@/test-utils/i18n';
import { WelcomeBanner } from '../WelcomeBanner';
import { screen } from '@testing-library/react';

const mockMessagesEn = {
  nav: { home: 'Home', about: 'About', contact: 'Contact' },
  common: {
    greeting: 'Hello, {name}!',
    itemCount: '{count, plural, one {# item} other {# items}}',
    terms: 'Please read our <link>terms of service</link>.',
  },
};

const mockMessagesDe = {
  nav: { home: 'Startseite', about: 'Über uns', contact: 'Kontakt' },
  common: {
    greeting: 'Hallo, {name}!',
    itemCount: '{count, plural, one {# Element} other {# Elemente}}',
    terms: 'Bitte lesen Sie unsere <link>Nutzungsbedingungen</link>.',
  },
};

describe('WelcomeBanner', () => {
  it('renders English greeting', () => {
    renderWithI18n(<WelcomeBanner name="Alice" />, {
      locale: 'en',
      messages: mockMessagesEn,
    });

    expect(screen.getByText('Hello, Alice!')).toBeInTheDocument();
    expect(screen.getByText('Home')).toBeInTheDocument();
  });

  it('renders German greeting', () => {
    renderWithI18n(<WelcomeBanner name="Alice" />, {
      locale: 'de',
      messages: mockMessagesDe,
    });

    expect(screen.getByText('Hallo, Alice!')).toBeInTheDocument();
    expect(screen.getByText('Startseite')).toBeInTheDocument();
  });

  it('pluralizes correctly', () => {
    renderWithI18n(<WelcomeBanner name="Bob" count={1} />, {
      locale: 'en',
      messages: mockMessagesEn,
    });
    expect(screen.getByText('1 item')).toBeInTheDocument();
  });

  it('matches locale snapshot', () => {
    const { container } = renderWithI18n(<WelcomeBanner name="World" />, {
      locale: 'en',
      messages: mockMessagesEn,
    });
    expect(container).toMatchSnapshot();
  });
});
```

## 12. Anti-Patterns

### Hardcoding strings in components

```tsx
// BAD — hardcoded English, impossible to translate
export function BadNav() {
  return (
    <nav>
      <a href="/">Home</a>
      <a href="/about">About Us</a>
      <a href="/contact">Get in Touch</a>
    </nav>
  );
}

// GOOD — translations driven from messages
export function GoodNav() {
  const t = useTranslations('nav');
  return (
    <nav>
      <Link href="/">{t('home')}</Link>
      <Link href="/about">{t('about')}</Link>
      <Link href="/contact">{t('contact')}</Link>
    </nav>
  );
}
```

### Not using Intl.* for dates and numbers

```tsx
// BAD — hardcoded formatting, breaks for non-English locales
function BadPrice({ amount }: { amount: number }) {
  return <span>${amount.toFixed(2)}</span>;
  // Always shows "$" regardless of locale; German users expect "1.234,56 €"
}

// GOOD — locale-aware formatting
function GoodPrice({ amount, currency }: { amount: number; currency: string }) {
  const format = useFormatter();
  return (
    <span>{format.number(amount, { style: 'currency', currency })}</span>
  );
}

// BAD — raw date string
function BadDate({ date }: { date: Date }) {
  return <time>{date.toString()}</time>;
  // Outputs browser-locale string, ignores app locale
}

// GOOD
function GoodDate({ date }: { date: Date }) {
  const format = useFormatter();
  return (
    <time dateTime={date.toISOString()}>
      {format.dateTime(date, { dateStyle: 'medium' })}
    </time>
  );
}
```

### Mixing translation keys with display strings

```tsx
// BAD — key leaks into UI if translation is missing
const label = t('nav.home') || 'nav.home';  // shows "nav.home" to users

// GOOD — always provide a fallback that looks like real content
// Or better: ensure all keys are present in every locale at build time
// Use the TypeScript types to catch missing translations early
```

### Forgetting RTL text alignment

```tsx
// BAD — hardcoded left alignment breaks RTL
<div className="text-left pl-4">Content</div>

// GOOD — logical properties that flip in RTL
<div className="text-start ps-4">Content</div>

// BAD — hardcoded directional icon
<ChevronRight className="ml-2" />

// GOOD — flip icon and margin for RTL
<ChevronRight className="ms-2 rtl:rotate-180" />
```

### Constructing translation keys dynamically

```tsx
// BAD — dynamic keys break static analysis and TypeScript type safety
const statusKey = `status.${orderStatus}`;
const label = t(statusKey); // TypeScript can't check this

// GOOD — explicit key map
const statusLabels = {
  pending: t('status.pending'),
  shipped: t('status.shipped'),
  delivered: t('status.delivered'),
} as const;

const label = statusLabels[orderStatus];
```

### Loading all locales on every request

```typescript
// BAD — loads all locale files into memory on every request
const allMessages = {
  en: await import('../messages/en.json'),
  de: await import('../messages/de.json'),
  ja: await import('../messages/ja.json'),
};
const messages = allMessages[locale];

// GOOD — only load the needed locale
const messages = (await import(`../messages/${locale}.json`)).default;
```

### Not scoping getTranslations with a namespace

```tsx
// BAD — loading the whole message tree; leaks keys across features
const t = await getTranslations({ locale });
t('nav.home'); // works but no scoping

// GOOD — namespace scopes to only what's needed
const t = await getTranslations({ locale, namespace: 'nav' });
t('home'); // cleaner API, better IDE support
```
